function generateAlerts(
  meetingsProcessed: number,
  successRate: number,
  brainLogCount: number,
  skillCount: number
): string[] {
  const alerts: AlertMessage[] = []
 
  if (successRate < 0.8) {
    alerts.push({
      level: 'warning',
      message: `Keyword dispatch success rate low: ${(successRate * 100).toFixed(1)}%`
    })
  }
 
  if (meetingsProcessed === 0) {
    alerts.push({
      level: 'info',
      message: 'No meetings processed yet'
    })
  }
 
  if (brainLogCount === 0) {
    alerts.push({
      level: 'warning',
      message: 'Company-brain not logging outcomes'
    })
  }
 
  if (skillCount === 0) {
    alerts.push({
      level: 'error',
      message: 'Skill registry not found or empty'
    })
  }
 
  return alerts.map(alert => `${ALERT_PREFIXES[alert.level]} ${alert.message}`)
}
